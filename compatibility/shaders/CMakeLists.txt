cmake_minimum_required(VERSION 3.16)
project(MavkShaderTool LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT MAVK_SPIRV_TOOLS_SOURCE)
  message(FATAL_ERROR "Pinned external SPIRV-Tools source required")
endif()
set(SPIRV_SKIP_TESTS ON CACHE BOOL "" FORCE)
set(SPIRV_SKIP_EXECUTABLES ON CACHE BOOL "" FORCE)
set(SPIRV_WERROR OFF CACHE BOOL "" FORCE)
add_subdirectory("${MAVK_SPIRV_TOOLS_SOURCE}" spirv-tools)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
configure_file("${MAVK_SPIRV_TOOLS_SOURCE}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/SPIRV-Tools-LICENSE.txt" COPYONLY)
configure_file("${MAVK_SPIRV_TOOLS_SOURCE}/external/spirv-headers/LICENSE" "${CMAKE_BINARY_DIR}/licenses/SPIRV-Headers-LICENSE.txt" COPYONLY)
add_library(mavk-shader-guards STATIC guards.cpp static_use.cpp)
target_include_directories(mavk-shader-guards PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_include_directories(mavk-shader-guards PRIVATE "${MAVK_SPIRV_TOOLS_SOURCE}" "${CMAKE_BINARY_DIR}/spirv-tools")
target_link_libraries(mavk-shader-guards PUBLIC SPIRV-Tools-opt)
target_compile_features(mavk-shader-guards PUBLIC cxx_std_17)
add_executable(mavk-shader-tool tool.cpp)
target_link_libraries(mavk-shader-tool PRIVATE mavk-shader-guards)
target_compile_features(mavk-shader-tool PRIVATE cxx_std_17)
add_executable(mavk-shader-guards-test ../test/shader-guards.cpp)
target_link_libraries(mavk-shader-guards-test PRIVATE mavk-shader-guards)
target_compile_features(mavk-shader-guards-test PRIVATE cxx_std_17)
add_executable(mavk-static-use-test ../test/static-use.cpp)
target_link_libraries(mavk-static-use-test PRIVATE mavk-shader-guards)
target_compile_features(mavk-static-use-test PRIVATE cxx_std_17)
add_library(mavk-cull-distance STATIC cull_distance.cpp)
target_include_directories(mavk-cull-distance PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_include_directories(mavk-cull-distance PRIVATE "${MAVK_SPIRV_TOOLS_SOURCE}/external/spirv-headers/include")
target_link_libraries(mavk-cull-distance PUBLIC SPIRV-Tools)
target_compile_features(mavk-cull-distance PUBLIC cxx_std_17)
add_executable(mavk-cull-tool ../test/cull-tool.cpp)
target_link_libraries(mavk-cull-tool PRIVATE mavk-cull-distance)
add_executable(mavk-cull-test ../test/cull-distance.cpp)
target_link_libraries(mavk-cull-test PRIVATE mavk-cull-distance)
